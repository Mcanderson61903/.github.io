def encrypt(text, shift):
    encrypted_text = ""  # Initialize an empty string to store the encrypted text
    for char in text:
        if char.isalpha():  # Check if the character is an alphabet letter
            shifted = ord(char) + shift  # Shift the character by the specified amount
            if char.islower():  # Check if the character is lowercase
                if shifted > ord('z'):  # Check if the shift exceeds 'z'
                    shifted -= 26  # Wrap around to 'a' if it exceeds 'z'
                elif shifted < ord('a'):  # Check if the shift goes below 'a'
                    shifted += 26  # Wrap around to 'z' if it goes below 'a'
            elif char.isupper():  # Check if the character is uppercase
                if shifted > ord('Z'):  # Check if the shift exceeds 'Z'
                    shifted -= 26  # Wrap around to 'A' if it exceeds 'Z'
                elif shifted < ord('A'):  # Check if the shift goes below 'A'
                    shifted += 26  # Wrap around to 'Z' if it goes below 'A'
            encrypted_text += chr(shifted)  # Append the shifted character to the encrypted text
        else:
            encrypted_text += char  # If the character is not an alphabet letter, keep it unchanged
    return encrypted_text  # Return the encrypted text


def decrypt(encrypted_text, shift):
    return encrypt(encrypted_text, -shift)  # Decrypting is just encrypting with a negative shift


# Example usage:
plaintext = input('Insert message to encrypt: ')  # Ask the user for input
shift = 3  # Set the shift value
encrypted_text = encrypt(plaintext, shift)  # Encrypt the plaintext
print("Encrypted:", encrypted_text)  # Print the encrypted text
decrypted_text = decrypt(encrypted_text, shift)  # Decrypt the encrypted text
print("Decrypted:", decrypted_text)  # Print the decrypted text
