import tkinter as tk
from tkinter import ttk, messagebox
from tkcalendar import Calendar


# Section 1: Calendar GUI
class CalendarApp:
    def __init__(self, root):
        self.root = root
        self.root.title("Calendar and To-Do List")

        self.setup_calendar()
        self.setup_todo_list()

    # Calendar Section
    def setup_calendar(self):
        calendar_frame = ttk.LabelFrame(self.root, text="Calendar")
        calendar_frame.grid(row=0, column=0, padx=10, pady=10)

        self.calendar = Calendar(calendar_frame, selectmode="day")
        self.calendar.pack(padx=10, pady=10)

        view_tasks_btn = ttk.Button(calendar_frame, text="View Tasks", command=self.view_tasks)
        view_tasks_btn.pack(pady=5)

    # To-Do List Section
    def setup_todo_list(self):
        todo_frame = ttk.LabelFrame(self.root, text="To-Do List")
        todo_frame.grid(row=0, column=1, padx=10, pady=10)

        self.task_listbox = tk.Listbox(todo_frame, height=15, width=30)
        self.task_listbox.pack(side=tk.LEFT, padx=5, pady=5)

        scrollbar = ttk.Scrollbar(todo_frame, orient="vertical", command=self.task_listbox.yview)
        scrollbar.pack(side=tk.RIGHT, fill="y")
        self.task_listbox.config(yscrollcommand=scrollbar.set)

        task_entry_frame = tk.Frame(todo_frame)
        task_entry_frame.pack(pady=10)

        self.task_entry = ttk.Entry(task_entry_frame, width=25)
        self.task_entry.grid(row=0, column=0, padx=5, pady=5)

        add_task_btn = ttk.Button(task_entry_frame, text="Add Task", command=self.add_task)
        add_task_btn.grid(row=0, column=1, padx=5)

        remove_task_btn = ttk.Button(todo_frame, text="Remove Task", command=self.remove_task)
        remove_task_btn.pack(pady=5)

        clear_tasks_btn = ttk.Button(todo_frame, text="Clear All", command=self.clear_tasks)
        clear_tasks_btn.pack(pady=5)

    # Event Handlers
    def add_task(self):
        task = self.task_entry.get().strip()
        if task:
            date = self.calendar.get_date()
            self.task_listbox.insert(tk.END, f"{date}: {task}")
            self.task_entry.delete(0, tk.END)
        else:
            messagebox.showwarning("Input Error", "Task cannot be empty!")

    def remove_task(self):
        selected_task_index = self.task_listbox.curselection()
        if selected_task_index:
            self.task_listbox.delete(selected_task_index)
        else:
            messagebox.showwarning("Selection Error", "No task selected to remove!")

    def clear_tasks(self):
        self.task_listbox.delete(0, tk.END)

    def view_tasks(self):
        selected_date = self.calendar.get_date()
        tasks_on_date = [task for task in self.task_listbox.get(0, tk.END) if task.startswith(selected_date)]

        if tasks_on_date:
            tasks_str = "\n".join(tasks_on_date)
            messagebox.showinfo(f"Tasks on {selected_date}", tasks_str)
        else:
            messagebox.showinfo(f"Tasks on {selected_date}", "No tasks found!")


if __name__ == "__main__":
    root = tk.Tk()
    app = CalendarApp(root)
    root.mainloop()
